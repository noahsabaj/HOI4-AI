cmake_minimum_required(VERSION 3.25)
project(hoi4_arena_bridge LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The bridge transport targets Windows")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(JSON_INCLUDE_DIR "" CACHE PATH "nlohmann-json 3.12.0 include directory")
if(NOT EXISTS "${JSON_INCLUDE_DIR}/nlohmann/json.hpp")
  message(FATAL_ERROR "Install .[native] and set JSON_INCLUDE_DIR to nlohmann_json/include")
endif()
add_executable(hoi4_bridge_probe src/probe.cpp)
target_include_directories(hoi4_bridge_probe PRIVATE "${JSON_INCLUDE_DIR}" include)
target_compile_definitions(hoi4_bridge_probe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_link_libraries(hoi4_bridge_probe PRIVATE advapi32)

# Explicitly a probe DLL: it has no game-state offsets or command hooks.
add_library(hoi4_bridge_inprocess_probe SHARED src/probe.cpp)
target_compile_features(hoi4_bridge_inprocess_probe PRIVATE cxx_std_20)
target_include_directories(hoi4_bridge_inprocess_probe PRIVATE "${JSON_INCLUDE_DIR}")
target_compile_definitions(hoi4_bridge_inprocess_probe PRIVATE HOI4_IN_PROCESS_PROBE
  NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE
  ARENA_SUPPORTED_EXECUTABLE_SHA256="dbc9744158e2e15c4182f04c868e9e697fa917a339072e831c5cde12d4563a76")
target_compile_options(hoi4_bridge_inprocess_probe PRIVATE /W4 /WX /EHsc)
target_link_libraries(hoi4_bridge_inprocess_probe PRIVATE advapi32 bcrypt)
if(MSVC)
  target_compile_options(hoi4_bridge_probe PRIVATE /W4 /WX /EHsc)
endif()
